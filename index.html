// Configurazione Sanity const PROJECT_ID = 'o0afaobs'; const DATASET = 'production'; // Questa query chiede a Sanity tutti i documenti di tipo "annuncio" ordinati per data const QUERY = encodeURIComponent('*[_type == "annuncio"] | order(data desc)'); const URL = `https://${PROJECT_ID}.api.sanity.io/v2021-10-21/data/query/${DATASET}?query=${QUERY}`; async function caricaBacheca() { try { const response = await fetch(URL); const { result } = await response.json(); const container = document.getElementById('bacheca-container'); // Assicurati di avere questo ID nel tuo HTML container.innerHTML = result.map(post => `

${post.titolo}

${new Date(post.data).toLocaleDateString('it-IT')}

${post.testo}

`).join(''); } catch (error) { console.error("Errore caricamento Sanity:", error); } } caricaBacheca();
